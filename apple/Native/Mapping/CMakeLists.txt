cmake_minimum_required(VERSION 3.22)
project(R2CMappingNative LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
 set(IOS_DEPLOYMENT_TARGET "${CMAKE_OSX_DEPLOYMENT_TARGET}")
endif()
set(EIGEN_INCLUDE_DIR "${OPENMVG_SOURCE}/src/third_party/eigen" CACHE PATH "" FORCE)
set(MINIGLOG ON CACHE BOOL "" FORCE)
set(OPENMP OFF CACHE BOOL "" FORCE)
set(EIGENSPARSE ON CACHE BOOL "" FORCE)
set(SUITESPARSE OFF CACHE BOOL "" FORCE)
set(CXSPARSE OFF CACHE BOOL "" FORCE)
set(LAPACK OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENMVG_SOURCE}/src/third_party/ceres-solver" ceres)
set_property(TARGET openMVG_ceres PROPERTY CXX_STANDARD 14)
set(BUILD_LIST "core,imgproc,features2d,calib3d,flann" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_PERF_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_opencv_apps OFF CACHE BOOL "" FORCE)
set(BUILD_JAVA OFF CACHE BOOL "" FORCE)
set(WITH_LAPACK OFF CACHE BOOL "" FORCE)
set(WITH_IPP OFF CACHE BOOL "" FORCE)
set(WITH_OPENCL OFF CACHE BOOL "" FORCE)
set(WITH_OPENEXR OFF CACHE BOOL "" FORCE)
set(WITH_ITT OFF CACHE BOOL "" FORCE)
set(WITH_PROTOBUF OFF CACHE BOOL "" FORCE)
set(OPENCV_ENABLE_NONFREE OFF CACHE BOOL "" FORCE)
add_subdirectory("${OPENCV_SOURCE}" opencv)
add_library(R2CMappingNative STATIC "${R2C_ROOT}/native/mapping/SurfaceCore.cpp" "${R2C_ROOT}/native/mapping/Reconstruction.cpp")
target_include_directories(R2CMappingNative PUBLIC "${R2C_ROOT}/native/mapping/include" PRIVATE
 "${R2C_ROOT}/native/mapping" "${EIGEN_INCLUDE_DIR}" "${OPENMVG_SOURCE}/src/third_party/ceres-solver/include"
 "${OPENMVG_SOURCE}/src/third_party/ceres-solver/internal/ceres/miniglog" "${CMAKE_CURRENT_BINARY_DIR}/ceres/config"
 "${OPENCV_SOURCE}/modules/core/include" "${OPENCV_SOURCE}/modules/imgproc/include"
 "${OPENCV_SOURCE}/modules/features2d/include" "${OPENCV_SOURCE}/modules/calib3d/include" "${OPENCV_SOURCE}/modules/flann/include" "${CMAKE_CURRENT_BINARY_DIR}/opencv" "${CMAKE_CURRENT_BINARY_DIR}")
target_link_libraries(R2CMappingNative PRIVATE openMVG_ceres opencv_core opencv_imgproc opencv_features2d opencv_calib3d opencv_flann)
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
 add_executable(R2CMappingNativeTests "${R2C_ROOT}/native/mapping/NativeTests.cpp")
 target_link_libraries(R2CMappingNativeTests PRIVATE R2CMappingNative)
 target_compile_options(R2CMappingNativeTests PRIVATE -UNDEBUG)
endif()
if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
 add_executable(R2CMappingPipelineTests "${R2C_ROOT}/native/mapping/PipelineTests.cpp")
 target_link_libraries(R2CMappingPipelineTests PRIVATE R2CMappingNative)
 target_include_directories(R2CMappingPipelineTests PRIVATE "${OPENCV_SOURCE}/modules/core/include" "${OPENCV_SOURCE}/modules/features2d/include" "${OPENCV_SOURCE}/modules/flann/include" "${CMAKE_CURRENT_BINARY_DIR}")
 target_compile_options(R2CMappingPipelineTests PRIVATE -UNDEBUG)
endif()

if(NOT CMAKE_SYSTEM_NAME STREQUAL "iOS")
 add_executable(R2CSurfaceRefinementTests "${R2C_ROOT}/native/mapping/SurfaceRefinementTests.cpp")
 target_compile_options(R2CSurfaceRefinementTests PRIVATE -UNDEBUG)
endif()
